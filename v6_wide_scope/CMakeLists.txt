cmake_minimum_required(VERSION 3.16)
project(causal_index_v5_boundary_join LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_subdirectory(
  ${CMAKE_CURRENT_SOURCE_DIR}/../v4_full_lattice
  ${CMAKE_CURRENT_BINARY_DIR}/v4_full_lattice
  EXCLUDE_FROM_ALL
)

add_library(causal_index_v5_boundary_join
  src/CausalAdjustment.cpp
  src/AdjustmentScopePlanner.cpp
  src/BoundaryJoinIndex.cpp
  src/BoundaryRequirementPlanner.cpp
  src/ParentTreePlanner.cpp
  src/Snapshot.cpp
)

target_include_directories(causal_index_v5_boundary_join PUBLIC include)
target_link_libraries(causal_index_v5_boundary_join PUBLIC causal_index_v4)
target_compile_options(causal_index_v5_boundary_join PRIVATE
  -Wall -Wextra -Wpedantic
)

add_executable(causal_index_v5_boundary_join_tests
  tests/boundary_join_tests.cpp
)
target_link_libraries(causal_index_v5_boundary_join_tests
  PRIVATE causal_index_v5_boundary_join
)
target_compile_options(causal_index_v5_boundary_join_tests PRIVATE -UNDEBUG)

add_executable(wide_scope_tests tests/wide_scope_tests.cpp)
target_link_libraries(wide_scope_tests PRIVATE causal_index_v5_boundary_join)
target_compile_options(wide_scope_tests PRIVATE -UNDEBUG)

add_executable(causal_index_v5_boundary_benchmark
  benchmarks/boundary_benchmark.cpp
)
target_link_libraries(causal_index_v5_boundary_benchmark
  PRIVATE causal_index_v5_boundary_join
)
target_compile_options(causal_index_v5_boundary_benchmark PRIVATE
  -Wall -Wextra -Wpedantic
)

enable_testing()
add_executable(causal_index_online_worker benchmarks/online_worker.cpp)
add_executable(materialize_online_boundaries benchmarks/materialize_online_boundaries.cpp)
target_link_libraries(materialize_online_boundaries PRIVATE causal_index_v5_boundary_join)
target_link_libraries(causal_index_online_worker PRIVATE causal_index_v5_boundary_join)
add_executable(causal_online_join_worker benchmarks/online_join_worker.cpp)
target_link_libraries(causal_online_join_worker PRIVATE causal_index_v5_boundary_join)
add_executable(full_join_rollup_tests tests/full_join_rollup_tests.cpp)
target_link_libraries(full_join_rollup_tests PRIVATE causal_index_v4)
target_compile_options(full_join_rollup_tests PRIVATE -UNDEBUG)
add_test(NAME full_join_rollup_tests COMMAND full_join_rollup_tests)
add_executable(streaming_full_join_tests tests/streaming_full_join_tests.cpp)
target_link_libraries(streaming_full_join_tests PRIVATE causal_index_v5_boundary_join)
target_compile_options(streaming_full_join_tests PRIVATE -UNDEBUG)
add_test(NAME streaming_full_join_tests COMMAND streaming_full_join_tests)
add_executable(scale_sampling_tests tests/scale_sampling_tests.cpp)
target_link_libraries(scale_sampling_tests PRIVATE causal_index_v5_boundary_join)
target_compile_options(scale_sampling_tests PRIVATE -UNDEBUG)
add_test(NAME scale_sampling_tests COMMAND scale_sampling_tests)
add_executable(join_metrics_tests tests/join_metrics_tests.cpp)
target_link_libraries(join_metrics_tests PRIVATE causal_index_v4)
target_compile_options(join_metrics_tests PRIVATE -UNDEBUG)
add_test(NAME join_metrics_tests COMMAND join_metrics_tests)
add_test(NAME wide_scope_tests COMMAND wide_scope_tests)
add_test(
  NAME causal_index_v5_boundary_join_tests
  COMMAND causal_index_v5_boundary_join_tests
)
